cmake_minimum_required(VERSION 3.25)
project(AegiNextAudio VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(AEGINEXT_FFMPEG_ROOT "" CACHE PATH "Pinned FFmpeg shared SDK")
set(AEGINEXT_SDL_ROOT "" CACHE PATH "SDL3 development SDK")
set(AEGINEXT_NATIVE_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/lib" CACHE PATH "Native output")
if(NOT AEGINEXT_FFMPEG_ROOT OR NOT AEGINEXT_SDL_ROOT)
    message(FATAL_ERROR "Explicit AEGINEXT_FFMPEG_ROOT and AEGINEXT_SDL_ROOT are required")
endif()
unset(FFMPEG_INCLUDE_DIR CACHE)
unset(SDL3_DIR CACHE)
find_path(FFMPEG_INCLUDE_DIR libavformat/avformat.h PATHS "${AEGINEXT_FFMPEG_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_package(SDL3 3.4.16 EXACT CONFIG REQUIRED PATHS "${AEGINEXT_SDL_ROOT}" "${AEGINEXT_SDL_ROOT}/lib/cmake/SDL3" NO_DEFAULT_PATH)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/AegiNext.Media/Probing/ffmpeg-toolchain.json" MANIFEST)
set(VERSION_HEADER "#pragma once\n")
foreach(COMPONENT avformat avcodec avutil swresample)
    unset(FFMPEG_${COMPONENT}_LIBRARY CACHE)
    find_library(FFMPEG_${COMPONENT}_LIBRARY NAMES ${COMPONENT} ${COMPONENT}.dll PATHS "${AEGINEXT_FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    string(JSON VERSION GET "${MANIFEST}" libraries "lib${COMPONENT}")
    string(REPLACE "." ";" PARTS "${VERSION}")
    list(GET PARTS 0 MAJOR)
    list(GET PARTS 1 MINOR)
    list(GET PARTS 2 MICRO)
    math(EXPR NUMBER "(${MAJOR} << 16) | (${MINOR} << 8) | ${MICRO}")
    string(TOUPPER "${COMPONENT}" UPPER)
    string(APPEND VERSION_HEADER "#define AN_EXPECTED_${UPPER} ${NUMBER}\n")
    list(APPEND AUDIO_LIBRARIES "${FFMPEG_${COMPONENT}_LIBRARY}")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/audio_versions.h" "${VERSION_HEADER}")
add_library(aeginext_audio SHARED src/audio.cpp src/audio_output_timeline.cpp)
if(APPLE)
    target_sources(aeginext_audio PRIVATE src/core_audio_output.cpp)
    target_link_libraries(aeginext_audio PRIVATE "-framework AudioToolbox" "-framework CoreAudio" "-framework CoreFoundation")
elseif(WIN32)
    target_sources(aeginext_audio PRIVATE src/wasapi_audio_output.cpp)
    target_link_libraries(aeginext_audio PRIVATE ole32 uuid ksuser avrt user32)
endif()
target_include_directories(aeginext_audio PUBLIC include PRIVATE "${FFMPEG_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(aeginext_audio PRIVATE AEGINEXT_AUDIO_BUILD)
target_link_libraries(aeginext_audio PRIVATE ${AUDIO_LIBRARIES} SDL3::SDL3)
set_target_properties(aeginext_audio PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
    LIBRARY_OUTPUT_DIRECTORY "${AEGINEXT_NATIVE_OUTPUT_DIR}" RUNTIME_OUTPUT_DIRECTORY "${AEGINEXT_NATIVE_OUTPUT_DIR}"
    ARCHIVE_OUTPUT_DIRECTORY "${AEGINEXT_NATIVE_OUTPUT_DIR}"
    BUILD_RPATH "${AEGINEXT_FFMPEG_ROOT}/lib;${AEGINEXT_SDL_ROOT}/lib")
if(WIN32)
    set_target_properties(aeginext_audio PROPERTIES PREFIX "")
endif()
if(MSVC)
    target_compile_options(aeginext_audio PRIVATE /W4 /WX)
else()
    target_compile_options(aeginext_audio PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(MINGW)
    target_link_options(aeginext_audio PRIVATE -static-libgcc -static-libstdc++)
    target_link_libraries(aeginext_audio PRIVATE -Wl,-Bstatic winpthread -Wl,-Bdynamic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(aeginext_audio_tests Tests/audio_tests.cpp src/audio_output_timeline.cpp)
    target_include_directories(aeginext_audio_tests PRIVATE src)
    target_link_libraries(aeginext_audio_tests PRIVATE aeginext_audio SDL3::SDL3)
    set_target_properties(aeginext_audio_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${AEGINEXT_NATIVE_OUTPUT_DIR}"
        BUILD_RPATH "${AEGINEXT_NATIVE_OUTPUT_DIR};${AEGINEXT_FFMPEG_ROOT}/lib;${AEGINEXT_SDL_ROOT}/lib")
    if(MINGW)
        target_link_options(aeginext_audio_tests PRIVATE -static-libgcc -static-libstdc++)
        target_link_libraries(aeginext_audio_tests PRIVATE -Wl,-Bstatic winpthread -Wl,-Bdynamic)
    endif()
    add_test(NAME aeginext.audio.contracts COMMAND aeginext_audio_tests)
    set_tests_properties(aeginext.audio.contracts PROPERTIES ENVIRONMENT "SDL_AUDIO_DRIVER=dummy")
endif()
