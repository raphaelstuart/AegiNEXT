cmake_minimum_required(VERSION 3.25)
if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Product macOS deployment target")
endif()
project(AegiNextExport VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/W4 /WX)
else()
    add_compile_options(-Wall -Wextra -Werror)
endif()
set(AEGINEXT_FFMPEG_ROOT "" CACHE PATH "Pinned FFmpeg SDK")
set(AEGINEXT_NATIVE_OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/lib" CACHE PATH "Native output")
if(NOT AEGINEXT_FFMPEG_ROOT)
    message(FATAL_ERROR "Explicit AEGINEXT_FFMPEG_ROOT is required")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Export ABI requires a 64-bit process")
endif()
unset(FFMPEG_INCLUDE CACHE)
find_path(FFMPEG_INCLUDE libavutil/csp.h PATHS "${AEGINEXT_FFMPEG_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
foreach(COMPONENT avformat avcodec avutil swscale)
    unset(FFMPEG_${COMPONENT} CACHE)
    find_library(FFMPEG_${COMPONENT} NAMES ${COMPONENT} ${COMPONENT}.dll PATHS "${AEGINEXT_FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    list(APPEND FFMPEG_LIBRARIES "${FFMPEG_${COMPONENT}}")
endforeach()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/AegiNext.Media/Probing/ffmpeg-toolchain.json" MANIFEST)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../src/AegiNext.Media/Probing/ffmpeg-toolchain.json")
set(VERSION_HEADER "#pragma once\n")
foreach(COMPONENT avformat avcodec avutil swscale)
    string(JSON VERSION GET "${MANIFEST}" libraries "lib${COMPONENT}")
    string(REPLACE "." ";" PARTS "${VERSION}")
    list(GET PARTS 0 MAJOR)
    list(GET PARTS 1 MINOR)
    list(GET PARTS 2 MICRO)
    math(EXPR VERSION_INT "(${MAJOR} << 16) | (${MINOR} << 8) | ${MICRO}")
    string(TOUPPER "${COMPONENT}" UPPER)
    string(APPEND VERSION_HEADER "#define AN_EXPORT_${UPPER} ${VERSION_INT}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/export_versions.h" "${VERSION_HEADER}")
add_library(aeginext_export SHARED src/export_api.cpp src/color_pipeline.cpp)
target_include_directories(aeginext_export PUBLIC include PRIVATE src "${FFMPEG_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(aeginext_export PRIVATE ${FFMPEG_LIBRARIES})
target_compile_definitions(aeginext_export PRIVATE AEGINEXT_EXPORT_BUILD)
set_target_properties(aeginext_export PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${AEGINEXT_NATIVE_OUTPUT_DIR}" RUNTIME_OUTPUT_DIRECTORY "${AEGINEXT_NATIVE_OUTPUT_DIR}" CXX_VISIBILITY_PRESET hidden BUILD_RPATH "${AEGINEXT_FFMPEG_ROOT}/lib")
if(WIN32)
    set_target_properties(aeginext_export PROPERTIES PREFIX "")
endif()
if(MINGW)
    target_link_options(aeginext_export PRIVATE -static-libgcc -static-libstdc++)
    target_link_libraries(aeginext_export PRIVATE -Wl,-Bstatic winpthread -Wl,-Bdynamic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(aeginext_export_tests Tests/color_tests.cpp src/color_pipeline.cpp)
    target_include_directories(aeginext_export_tests PRIVATE src "${FFMPEG_INCLUDE}")
    target_link_libraries(aeginext_export_tests PRIVATE ${FFMPEG_LIBRARIES})
    set_target_properties(aeginext_export_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${AEGINEXT_NATIVE_OUTPUT_DIR}" BUILD_RPATH "${AEGINEXT_FFMPEG_ROOT}/lib")
    add_test(NAME aeginext.export.colors COMMAND aeginext_export_tests)
endif()
if(BUILD_TESTING)
    add_executable(aeginext_export_api_tests Tests/api_tests.cpp)
    target_include_directories(aeginext_export_api_tests PRIVATE include)
    target_link_libraries(aeginext_export_api_tests PRIVATE aeginext_export)
    set_target_properties(aeginext_export_api_tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${AEGINEXT_NATIVE_OUTPUT_DIR}" BUILD_RPATH "${AEGINEXT_NATIVE_OUTPUT_DIR};${AEGINEXT_FFMPEG_ROOT}/lib")
    add_test(NAME aeginext.export.api COMMAND aeginext_export_api_tests)
    add_executable(aeginext_export_fixture Tests/export_fixture.cpp)
    target_include_directories(aeginext_export_fixture PRIVATE include)
    target_link_libraries(aeginext_export_fixture PRIVATE aeginext_export)
    set_target_properties(aeginext_export_fixture PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${AEGINEXT_NATIVE_OUTPUT_DIR}" BUILD_RPATH "${AEGINEXT_NATIVE_OUTPUT_DIR};${AEGINEXT_FFMPEG_ROOT}/lib")
endif()
