get_filename_component(CORE_MANIFEST_PATH "${CMAKE_CURRENT_LIST_DIR}/../../src/AegiNext.Media/Probing/ffmpeg-toolchain.json" REALPATH)
file(READ "${CORE_MANIFEST_PATH}" CORE_MANIFEST)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CORE_MANIFEST_PATH}")

set(CORE_VERSION_HEADER "#pragma once\n")
foreach(COMPONENT avformat avcodec avutil swscale)
    string(JSON VERSION GET "${CORE_MANIFEST}" libraries "lib${COMPONENT}")
    string(REPLACE "." ";" PARTS "${VERSION}")
    list(GET PARTS 0 MAJOR)
    list(GET PARTS 1 MINOR)
    list(GET PARTS 2 MICRO)
    math(EXPR NUMBER "(${MAJOR} << 16) | (${MINOR} << 8) | ${MICRO}")
    string(TOUPPER "${COMPONENT}" UPPER)
    string(APPEND CORE_VERSION_HEADER "#define CORE_EXPECTED_${UPPER} ${NUMBER}\n")
endforeach()
string(JSON COUNT LENGTH "${CORE_MANIFEST}" acceptedVersionStrings)
math(EXPR LAST "${COUNT} - 1")
string(APPEND CORE_VERSION_HEADER "inline constexpr const char *CORE_RELEASES[] = {\n")
foreach(INDEX RANGE ${LAST})
    string(JSON RELEASE GET "${CORE_MANIFEST}" acceptedVersionStrings ${INDEX})
    if(NOT RELEASE MATCHES "^[A-Za-z0-9._+-]+$")
        message(FATAL_ERROR "Invalid FFmpeg release string.")
    endif()
    string(APPEND CORE_VERSION_HEADER "\"${RELEASE}\",\n")
endforeach()
string(APPEND CORE_VERSION_HEADER "};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/media_core_versions.h" "${CORE_VERSION_HEADER}")
add_library(aeginext_media_core STATIC media_core.cpp color_resolution.cpp source_alpha_compositor.cpp hardware_decode_policy.cpp)
target_include_directories(aeginext_media_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${FFMPEG_INCLUDE_DIR}" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(aeginext_media_core PUBLIC "${FFMPEG_avformat_LIBRARY}" "${FFMPEG_avcodec_LIBRARY}" "${FFMPEG_avutil_LIBRARY}" "${FFMPEG_swscale_LIBRARY}")
set_target_properties(aeginext_media_core PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(APPLE)
    target_link_libraries(aeginext_media_core PUBLIC "-framework VideoToolbox" "-framework CoreVideo" "-framework CoreMedia" "-framework CoreFoundation")
endif()
if(WIN32)
    target_sources(aeginext_media_core PRIVATE windows_vulkan_device.cpp)
    target_link_libraries(aeginext_media_core PUBLIC dxgi)
endif()
if(MSVC)
    target_compile_options(aeginext_media_core PRIVATE /W4 /WX)
else()
    target_compile_options(aeginext_media_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
